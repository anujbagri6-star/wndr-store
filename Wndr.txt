    // Category Filter Handler
    function filterCategory(cat) {
      document.querySelectorAll('.cat-btn').forEach(btn => {
        if(btn.dataset.cat === cat) {
          btn.className = "cat-btn px-4 py-2 bg-[#ccff00] text-black font-bold uppercase rounded-sm border border-[#ccff00]";
        } else {
          btn.className = "cat-btn px-4 py-2 bg-[#121216] text-neutral-300 font-bold uppercase rounded-sm border border-[#23232b] hover:border-[#ccff00]";
        }
      });
      renderProducts(cat);
    }

    // Cart Management
    function addToCart(productId) {
      const item = PRODUCTS.find(p => p.id === productId);
      const existing = cart.find(i => i.id === productId);

      if (existing) {
        existing.quantity += 1;
      } else {
        cart.push({ ...item, quantity: 1 });
      }

      updateCartUI();
      openCartDrawer();
    }

    function changeQuantity(id, delta) {
      const item = cart.find(i => i.id === id);
      if (!item) return;

      item.quantity += delta;
      if (item.quantity <= 0) {
        cart = cart.filter(i => i.id !== id);
      }
      updateCartUI();
    }

    function updateCartUI() {
      // Counter
      const count = cart.reduce((acc, item) => acc + item.quantity, 0);
      document.getElementById('cart-counter').innerText = count;

      // Cart Items HTML
      const itemsContainer = document.getElementById('cart-items');
      if (cart.length === 0) {
        itemsContainer.innerHTML = `
          <div class="h-full flex flex-col items-center justify-center text-center text-neutral-500 py-12">
            <i data-lucide="shopping-bag" class="w-12 h-12 mb-3 stroke-1"></i>
            <p class="font-mono text-xs uppercase">Your cart is currently empty.</p>
          </div>
        `;
      } else {
        itemsContainer.innerHTML = cart.map(item => `
          <div class="flex gap-4 p-3 bg-[#15151c] border border-[#23232b] rounded-sm items-center">
            <img src="${item.image}" class="w-16 h-16 object-cover rounded-sm" />
            <div class="flex-1">
              <h4 class="font-heading text-xs font-bold leading-tight mb-1 text-white">${item.name}</h4>
              <div class="font-mono text-xs text-[#ccff00]">₹${item.price.toLocaleString('en-IN')}</div>
              <div class="flex items-center gap-3 mt-2">
                <button onclick="changeQuantity(${item.id}, -1)" class="w-5 h-5 bg-[#23232b] text-white flex items-center justify-center font-mono text-xs hover:bg-[#ccff00] hover:text-black">-</button>
                <span class="font-mono text-xs font-bold">${item.quantity}</span>
                <button onclick="changeQuantity(${item.id}, 1)" class="w-5 h-5 bg-[#23232b] text-white flex items-center justify-center font-mono text-xs hover:bg-[#ccff00] hover:text-black">+</button>
              </div>
            </div>
          </div>
        `).join('');
      }

      // Financials
      const subtotal = cart.reduce((acc, i) => acc + (i.price * i.quantity), 0);
      const discount = (subtotal * discountPercent) / 100;
      const isFreeDelivery = subtotal >= 1499 || subtotal === 0;
      const shipping = isFreeDelivery ? 0 : 99;
      const total = Math.max(0, subtotal - discount + shipping);

      document.getElementById('cart-subtotal').innerText = `₹${subtotal.toLocaleString('en-IN')}`;
      document.getElementById('cart-discount').innerText = `- ₹${discount.toLocaleString('en-IN')}`;
      document.getElementById('cart-shipping').innerText = isFreeDelivery ? "FREE" : `₹${shipping}`;
      document.getElementById('cart-total').innerText = `₹${total.toLocaleString('en-IN')}`;

      // Free shipping progress bar
      const bar = document.getElementById('shipping-bar');
      const label = document.getElementById('shipping-label');
      if (subtotal >= 1499) {
        bar.style.width = '100%';
        label.innerText = '⚡ FREE Express Shipping Unlocked!';
        label.className = 'text-[#ccff00] font-bold';
      } else {
        const pct = Math.min(100, (subtotal / 1499) * 100);
        bar.style.width = `${pct}%`;
        label.innerText = `Add ₹${(1499 - subtotal).toLocaleString('en-IN')} for Free Delivery`;
        label.className = 'text-neutral-300';
      }

      lucide.createIcons();
    }

    // Coupon logic
    function applyCoupon() {
      const code = document.getElementById('coupon-input').value.trim().toUpperCase();
      if (code === 'WNDR20') {
        discountPercent = 20;
        alert("20% Street Essentials discount applied!");
      } else if (code === 'WNDR10') {
        discountPercent = 10;
        alert("10% VIP discount applied!");
      } else {
        alert("Invalid coupon code. Try 'WNDR20'");
      }
      updateCartUI();
    }

    // Drawer Controls
    function toggleCart() {
      const drawer = document.getElementById('cart-drawer');
      const backdrop = document.getElementById('cart-backdrop');
      const panel = document.getElementById('cart-panel');

      if (panel.classList.contains('translate-x-full')) {
        openCartDrawer();
      } else {
        drawer.classList.add('pointer-events-none');
        backdrop.classList.add('opacity-0');
        backdrop.classList.add('pointer-events-none');
        panel.classList.add('translate-x-full');
      }
    }

    function openCartDrawer() {
      const drawer = document.getElementById('cart-drawer');
      const backdrop = document.getElementById('cart-backdrop');
      const panel = document.getElementById('cart-panel');

      drawer.classList.remove('pointer-events-none');
      backdrop.classList.remove('opacity-0');
      backdrop.classList.remove('pointer-events-none');
      panel.classList.remove('translate-x-full');
    }

    function checkout() {
      if (cart.length === 0) {
        alert("Your cart is empty!");
        return;
      }
      alert("Redirecting to Razorpay / Stripe Secure Checkout...");
    }

    function handleSubscribe(e) {
      e.preventDefault();
      document.getElementById('sub-msg').classList.remove('hidden');
    }
